// ====== HTTP helpers ======
async function supabaseRequest(table, method, options = {}) {
const url = `${SUPABASE\_URL}/rest/v1/${table}`;
const headers = {
'apikey': SUPABASE\_ANON\_KEY,
'Authorization': `Bearer ${SUPABASE\_ANON\_KEY}`,
'Content-Type': 'application/json',
'Prefer': options.prefer || 'return=representation',
};
let queryParams = '';
if (options.select) queryParams += `?select=${encodeURIComponent(options.select)}`;
if (options.eq) {
for (const [col, val] of Object.entries(options.eq)) {
queryParams += `${queryParams ? '&' : '?'}${col}=eq.${encodeURIComponent(val)}`;
}
}
if (options.order) queryParams += `${queryParams ? '&' : '?'}order=${encodeURIComponent(options.order)}`;
if (options.limit != null) queryParams += `${queryParams ? '&' : '?'}limit=${options.limit}`;
const controller = new AbortController();
const timeout = setTimeout(() => controller.abort(), SYNC\_TIMEOUT\_MS);
try {
const res = await fetch(url + queryParams, {
method,
headers,
body: options.body ? JSON.stringify(options.body) : undefined,
signal: controller.signal,
});
clearTimeout(timeout);
if (!res.ok) {
const errText = await res.text();
throw new Error(`Supabase ${method} ${table} failed: ${res.status} ${errText}`);
}
const ct = res.headers.get('content-type') || '';
if (ct.includes('json')) return await res.json();
return null;
} catch (e) {
clearTimeout(timeout);
throw e;
}
}
async function supabaseInsert(table, records) {
const arr = Array.isArray(records) ? records : [records];
return supabaseRequest(table, 'POST', { body: arr });
}
async function supabaseSelect(table, options = {}) {

}
async function supabaseDelete(table, options = {}) {
return supabaseRequest(table, 'DELETE', { prefer: 'return=minimal', ...options });
}
